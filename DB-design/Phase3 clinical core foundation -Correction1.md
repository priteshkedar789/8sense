-- =============================================================================
-- [FIX-P1] validate_case_role_scope_ref() — remove syntax errors
-- =============================================================================

CREATE OR REPLACE FUNCTION validate_case_role_scope_ref()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE
    v_scope_code    TEXT;
    v_patient_id    UUID;
    v_dept_id       UUID;
    v_prog_id       UUID;
BEGIN
    SELECT csl.code INTO v_scope_code
    FROM case_scope_levels csl
    WHERE csl.id = NEW.scope_level_id;

    -- patient scope
    IF v_scope_code = 'patient' THEN
        IF NEW.scope_ref_id IS NULL THEN
            RAISE EXCEPTION
                'case_role_assignment: scope_level=patient requires scope_ref_id (patient.id). '
                'assignment_id: %',
                NEW.id;
        END IF;
        SELECT id INTO v_patient_id
        FROM patients
        WHERE id = NEW.scope_ref_id
          AND institute_id = NEW.institute_id;
        IF NOT FOUND THEN
            RAISE EXCEPTION
                'case_role_assignment: scope_ref_id % is not a valid patient '
                'in institute %. scope_level=patient.',
                NEW.scope_ref_id, NEW.institute_id;
        END IF;

    -- department scope
    ELSIF v_scope_code = 'department' THEN
        IF NEW.scope_ref_id IS NULL THEN
            RAISE EXCEPTION
                'case_role_assignment: scope_level=department requires scope_ref_id (department.id). '
                'assignment_id: %',
                NEW.id;
        END IF;
        SELECT id INTO v_dept_id
        FROM departments
        WHERE id = NEW.scope_ref_id
          AND institute_id = NEW.institute_id;
        IF NOT FOUND THEN
            RAISE EXCEPTION
                'case_role_assignment: scope_ref_id % is not a valid department '
                'in institute %. scope_level=department.',
                NEW.scope_ref_id, NEW.institute_id;
        END IF;

    -- program scope — UNBLOCKED [O2], therapy_programs exists
    ELSIF v_scope_code = 'program' THEN
        IF NEW.scope_ref_id IS NULL THEN
            RAISE EXCEPTION
                'case_role_assignment: scope_level=program requires scope_ref_id (therapy_program.id). '
                'assignment_id: %',
                NEW.id;
        END IF;
        SELECT id INTO v_prog_id
        FROM therapy_programs
        WHERE id = NEW.scope_ref_id
          AND institute_id = NEW.institute_id;
        IF NOT FOUND THEN
            RAISE EXCEPTION
                'case_role_assignment: scope_ref_id % is not a valid therapy_program '
                'in institute %. scope_level=program.',
                NEW.scope_ref_id, NEW.institute_id;
        END IF;

    ELSE
        RAISE EXCEPTION
            'case_role_assignment: unrecognised scope_level code %. '
            'Valid values: patient, department, program. '
            'assignment_id: %',
            v_scope_code, NEW.id;
    END IF;

    RETURN NEW;
END;
$$;

COMMENT ON FUNCTION validate_case_role_scope_ref() IS
    '[FIX-P1] Syntax errors corrected (trailing comma before semicolon removed). '
    '[O2] Full three-way scope validation: patient, department, program. '
    'All branches validate scope_ref_id against target table + institute boundary. '
    'Unrecognised scope codes fail loudly — future scope types cannot silently bypass.';


-- =============================================================================
-- [FIX-P2] Amendment chain: one amendment per original [D-P1]
-- =============================================================================
-- A tree amendment model implies governance failure — parallel corrections
-- create competing authoritative versions with no defined resolution.
-- Chain enforced: each original response can have exactly one amendment.
-- If a second correction is needed, it amends the first amendment.
-- =============================================================================

CREATE UNIQUE INDEX uq_fr_single_amendment
    ON form_responses(supersedes_response_id)
    WHERE supersedes_response_id IS NOT NULL;

COMMENT ON INDEX uq_fr_single_amendment IS
    '[FIX-P2] [D-P1] Enforces chain amendment model. '
    'Each original response can have exactly one amendment (one child). '
    'A second correction must amend the first amendment, not the original. '
    'Correction chain: original → amendment_1 → amendment_2 → ... '
    'Tree structure (multiple children per original) is structurally impossible. '
    'Audit traversal is linear — no reconciliation ambiguity.';


-- =============================================================================
-- [FIX-P3] program_goals composite FK: version must belong to same program
-- =============================================================================
-- UNIQUE (id, therapy_program_id) required on therapy_program_versions
-- to serve as FK target. Adding it now if not present.
-- =============================================================================

-- Enable composite FK target on therapy_program_versions
ALTER TABLE therapy_program_versions
    ADD CONSTRAINT uq_tpv_id_program
    UNIQUE (id, therapy_program_id);

-- Composite FK on program_goals: version must belong to the same program as goal
ALTER TABLE program_goals
    DROP CONSTRAINT IF EXISTS program_goals_program_version_id_fkey;

ALTER TABLE program_goals
    ADD CONSTRAINT fk_goal_version_program
    FOREIGN KEY (program_version_id, therapy_program_id)
    REFERENCES therapy_program_versions(id, therapy_program_id)
    DEFERRABLE INITIALLY DEFERRED;

COMMENT ON CONSTRAINT fk_goal_version_program ON program_goals IS
    '[FIX-P3] Composite FK ensures the program_version_id belongs to the same '
    'therapy_program as the goal. Prevents cross-program goal-version attachment '
    '(goal.therapy_program_id = A, goal.program_version_id → program B).';


-- =============================================================================
-- [FIX-P4] version_number auto-generated via trigger with FOR UPDATE [D-P3]
-- =============================================================================
-- Application never supplies version_number — trigger computes MAX+1.
-- FOR UPDATE on therapy_programs row serializes concurrent version inserts.
-- Guarantees: no gaps, no reversals, no concurrent collisions.
-- =============================================================================

-- Make version_number nullable so application can omit it on INSERT
-- Trigger will set it before the row lands
ALTER TABLE therapy_program_versions
    ALTER COLUMN version_number DROP NOT NULL;

CREATE OR REPLACE FUNCTION generate_program_version_number()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE
    v_max_version INTEGER;
BEGIN
    -- [FIX-P4] Lock the parent program row to serialize concurrent version inserts.
    -- This prevents two concurrent inserts from both reading the same MAX and
    -- then racing to insert the same version_number.
    PERFORM 1
    FROM therapy_programs
    WHERE id = NEW.therapy_program_id
    FOR UPDATE;

    SELECT COALESCE(MAX(version_number), 0) INTO v_max_version
    FROM therapy_program_versions
    WHERE therapy_program_id = NEW.therapy_program_id;

    NEW.version_number := v_max_version + 1;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_generate_program_version_number
    BEFORE INSERT ON therapy_program_versions
    FOR EACH ROW
    EXECUTE FUNCTION generate_program_version_number();

COMMENT ON TRIGGER trg_generate_program_version_number ON therapy_program_versions IS
    '[FIX-P4] [D-P3] Auto-generates monotonically increasing version_number. '
    'Acquires FOR UPDATE lock on therapy_programs row before computing MAX+1. '
    'Application must not supply version_number — trigger always overwrites it. '
    'Guarantees: strictly sequential, no gaps, no reversals, concurrency-safe.';

COMMENT ON FUNCTION generate_program_version_number() IS
    '[FIX-P4] [D-P3] Called BEFORE INSERT on therapy_program_versions. '
    'Sets NEW.version_number = MAX(version_number)+1 for this program. '
    'FOR UPDATE on parent program row serializes concurrent inserts. '
    'COALESCE(MAX, 0): first version for a program gets version_number = 1.';


-- =============================================================================
-- [FIX-P5] department_id added to event fact freeze [D-P2]
-- =============================================================================
-- department_id drives billing, reporting, and RLS scoping.
-- Retrospective reassignment rewrites revenue attribution and KPIs.
-- Controlled department transfers: Phase 4 with department_transfer_log.
-- =============================================================================

CREATE OR REPLACE FUNCTION enforce_therapy_program_event_freeze()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    -- Domain A freeze: identity facts immutable once program becomes active
    IF OLD.program_status IN ('active', 'paused', 'completed', 'discontinued') THEN
        IF NEW.patient_id               IS DISTINCT FROM OLD.patient_id               OR
           NEW.institute_id             != OLD.institute_id                            OR
           NEW.branch_id                IS DISTINCT FROM OLD.branch_id                OR
           NEW.department_id            IS DISTINCT FROM OLD.department_id            OR  -- [FIX-P5]
           NEW.primary_therapy_type_id  IS DISTINCT FROM OLD.primary_therapy_type_id  OR
           NEW.actual_start_date        IS DISTINCT FROM OLD.actual_start_date
        THEN
            RAISE EXCEPTION
                '[FIX-P5] therapy_program % has status=% — identity facts are frozen. '
                'patient_id, institute_id, branch_id, department_id, '
                'primary_therapy_type_id, and actual_start_date cannot change '
                'once the program is active. '
                'Use amendment workflow (supersedes_program_id) for corrections. '
                'Department transfers will be supported in Phase 4 via department_transfer_log.',
                OLD.id, OLD.program_status;
        END IF;
    END IF;

    -- Domain B freeze: completed/discontinued programs cannot be reopened
    IF OLD.program_status IN ('completed', 'discontinued') THEN
        IF NEW.program_status NOT IN ('completed', 'discontinued') THEN
            RAISE EXCEPTION
                'therapy_program % is % and cannot be reopened. '
                'Create a new program (with supersedes_program_id) to resume care.',
                OLD.id, OLD.program_status;
        END IF;
    END IF;

    RETURN NEW;
END;
$$;

COMMENT ON FUNCTION enforce_therapy_program_event_freeze() IS
    '[FIX-P5] department_id added to identity freeze. '
    'Domain A: patient, institute, branch, department, primary_type, actual_start_date '
    'frozen at active. department_id change after activation rewrites billing/reporting. '
    'Domain B: completed/discontinued programs cannot be reopened. '
    'Phase 4 will add department_transfer_log for controlled retrospective transfers.';


-- =============================================================================
-- [FIX-P6] therapy_program_type_map frozen at program activation [D-P5]
-- =============================================================================
-- Secondary therapy types affect reporting as much as primary type.
-- Changing types after activation causes the same revenue and reporting drift.
-- Map is frozen when the parent program reaches active status.
-- =============================================================================

CREATE OR REPLACE FUNCTION enforce_program_type_map_freeze()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE
    v_program_status TEXT;
BEGIN
    SELECT program_status INTO v_program_status
    FROM therapy_programs
    WHERE id = COALESCE(NEW.therapy_program_id, OLD.therapy_program_id);

    IF v_program_status IN ('active', 'paused', 'completed', 'discontinued') THEN
        IF TG_OP = 'DELETE' THEN
            RAISE EXCEPTION
                '[FIX-P6] Cannot remove therapy type mapping from program % '
                '— program is % (active or beyond). '
                'Therapy type assignments are frozen at activation.',
                OLD.therapy_program_id, v_program_status;
        END IF;
        IF TG_OP IN ('INSERT', 'UPDATE') THEN
            RAISE EXCEPTION
                '[FIX-P6] Cannot add or modify therapy type mapping on program % '
                '— program is % (active or beyond). '
                'Therapy type assignments are frozen at activation. '
                'Use amendment workflow (supersedes_program_id) for type changes.',
                NEW.therapy_program_id, v_program_status;
        END IF;
    END IF;

    RETURN COALESCE(NEW, OLD);
END;
$$;

CREATE TRIGGER trg_program_type_map_freeze
    BEFORE INSERT OR UPDATE OR DELETE ON therapy_program_type_map
    FOR EACH ROW
    EXECUTE FUNCTION enforce_program_type_map_freeze();

COMMENT ON TRIGGER trg_program_type_map_freeze ON therapy_program_type_map IS
    '[FIX-P6] [D-P5] Blocks INSERT, UPDATE, DELETE on type map once program is active. '
    'Secondary types affect reporting and billing attribution identically to primary type. '
    'Frozen at active (same boundary as department_id and primary_therapy_type_id). '
    'Amendment workflow required for type changes after activation.';


-- =============================================================================
-- [FIX-P7] therapy_program_assignments frozen on closed programs [D-P4]
-- =============================================================================
-- Changing provider assignments after a program is completed or discontinued
-- rewrites: who delivered care, provider productivity, payroll reconciliation,
-- and supervision audit trails. All of these are regulatory data.
-- =============================================================================

CREATE OR REPLACE FUNCTION enforce_program_assignment_freeze_on_close()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE
    v_program_status TEXT;
BEGIN
    SELECT program_status INTO v_program_status
    FROM therapy_programs
    WHERE id = COALESCE(NEW.therapy_program_id, OLD.therapy_program_id);

    IF v_program_status IN ('completed', 'discontinued') THEN
        IF TG_OP = 'DELETE' THEN
            RAISE EXCEPTION
                '[FIX-P7] Cannot remove provider assignment from program % '
                '— program is % (closed). '
                'Provider assignment history is frozen when a program closes.',
                OLD.therapy_program_id, v_program_status;
        END IF;
        IF TG_OP = 'INSERT' THEN
            RAISE EXCEPTION
                '[FIX-P7] Cannot add a provider assignment to program % '
                '— program is % (closed). '
                'No new assignments may be created on a closed program.',
                NEW.therapy_program_id, v_program_status;
        END IF;
        IF TG_OP = 'UPDATE' THEN
            RAISE EXCEPTION
                '[FIX-P7] Cannot modify a provider assignment on program % '
                '— program is % (closed). '
                'Provider assignments are immutable on closed programs. '
                'This preserves payroll reconciliation and supervision audit trails.',
                OLD.therapy_program_id, v_program_status;
        END IF;
    END IF;

    RETURN COALESCE(NEW, OLD);
END;
$$;

CREATE TRIGGER trg_program_assignment_freeze_on_close
    BEFORE INSERT OR UPDATE OR DELETE ON therapy_program_assignments
    FOR EACH ROW
    EXECUTE FUNCTION enforce_program_assignment_freeze_on_close();

COMMENT ON TRIGGER trg_program_assignment_freeze_on_close ON therapy_program_assignments IS
    '[FIX-P7] [D-P4] Blocks all writes (INSERT/UPDATE/DELETE) to provider assignments '
    'once the program is completed or discontinued. '
    'Protects: who delivered care, provider productivity reporting, '
    'payroll reconciliation, supervision audit trails. '
    'Active and paused programs: assignments still mutable (handover workflows allowed). '
    'Completed/discontinued: fully frozen.';
